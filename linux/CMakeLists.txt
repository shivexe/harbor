cmake_minimum_required(VERSION 3.22)
project(Harbor VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
find_package(OpenSSL REQUIRED)
find_package(qtermwidget6 REQUIRED)
find_library(QRENCODE_LIBRARY qrencode REQUIRED)
find_path(QRENCODE_INCLUDE_DIR qrencode.h REQUIRED)
add_library(harbor-core src/vault.cpp src/sync.cpp src/pairing_network.cpp)
target_include_directories(harbor-core PUBLIC src)
target_link_libraries(harbor-core PUBLIC Qt6::Network OpenSSL::Crypto)
add_executable(harbor src/main.cpp src/window.cpp src/terminal.cpp src/resources.qrc)
target_link_libraries(harbor PRIVATE harbor-core Qt6::Widgets qtermwidget6 ${QRENCODE_LIBRARY})
target_include_directories(harbor PRIVATE ${QRENCODE_INCLUDE_DIR})
add_executable(harbor-askpass src/askpass.cpp)
target_link_libraries(harbor-askpass PRIVATE Qt6::Network)
include(CTest)
if(BUILD_TESTING)
    add_executable(harbor-security-test tests/security.cpp)
    target_link_libraries(harbor-security-test PRIVATE harbor-core)
    add_dependencies(harbor-security-test harbor-askpass)
    target_compile_definitions(harbor-security-test PRIVATE HARBOR_VECTOR_FILE="${CMAKE_CURRENT_SOURCE_DIR}/../shared/protocol-vectors.json")
    add_test(NAME security COMMAND harbor-security-test)
    add_executable(harbor-sync-fixture tests/sync_fixture.cpp)
    target_link_libraries(harbor-sync-fixture PRIVATE harbor-core)
    add_executable(harbor-gui-fixture tests/gui_fixture.cpp src/window.cpp src/terminal.cpp src/resources.qrc)
    target_link_libraries(harbor-gui-fixture PRIVATE harbor-core Qt6::Widgets qtermwidget6 ${QRENCODE_LIBRARY})
    target_include_directories(harbor-gui-fixture PRIVATE ${QRENCODE_INCLUDE_DIR})
    add_executable(harbor-ui-fixture tests/ui_fixture.cpp src/window.cpp src/terminal.cpp src/resources.qrc)
    target_link_libraries(harbor-ui-fixture PRIVATE harbor-core Qt6::Widgets qtermwidget6 ${QRENCODE_LIBRARY})
    target_include_directories(harbor-ui-fixture PRIVATE ${QRENCODE_INCLUDE_DIR})
    add_executable(harbor-key-editor-test tests/key_editor.cpp src/window.cpp src/terminal.cpp src/resources.qrc)
    target_link_libraries(harbor-key-editor-test PRIVATE harbor-core Qt6::Widgets qtermwidget6 ${QRENCODE_LIBRARY})
    target_include_directories(harbor-key-editor-test PRIVATE ${QRENCODE_INCLUDE_DIR})
    add_executable(harbor-encrypted-connection-test tests/encrypted_connection.cpp src/terminal.cpp src/resources.qrc)
    target_link_libraries(harbor-encrypted-connection-test PRIVATE harbor-core Qt6::Widgets qtermwidget6)
    add_dependencies(harbor-encrypted-connection-test harbor-askpass)
    add_executable(harbor-connection-fixture tests/connection_fixture.cpp src/window.cpp src/terminal.cpp src/resources.qrc)
    target_link_libraries(harbor-connection-fixture PRIVATE harbor-core Qt6::Widgets qtermwidget6 ${QRENCODE_LIBRARY})
    target_include_directories(harbor-connection-fixture PRIVATE ${QRENCODE_INCLUDE_DIR})
endif()
install(TARGETS harbor harbor-askpass RUNTIME DESTINATION bin)
install(FILES harbor.desktop DESTINATION share/applications)
install(FILES assets/harbor.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION share/doc/harbor)
set(CPACK_PACKAGE_NAME harbor)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Harbor project")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Encrypted SSH workspace and desktop-to-Android sync")
set(CPACK_GENERATOR DEB)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6core6t64 (>= 6.10.2), libqt6gui6 (>= 6.10.2), libqt6widgets6 (>= 6.10.2), libqt6network6 (>= 6.10.2), libqtermwidget6-2 (>= 2.3), libssl3t64 (>= 3.5), libqrencode4, openssh-client, fontconfig, fonts-dejavu-core")
set(CPACK_DEBIAN_PACKAGE_SECTION net)
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE amd64)
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_PACKAGE_FILE_NAME "harbor-${PROJECT_VERSION}-linux-amd64")
include(CPack)
